cmake_minimum_required(VERSION 3.10)
project(portfolio_embedded_cpp LANGUAGES CXX)
enable_testing()
add_executable(core_tests cpp/tests/test.cpp)
target_include_directories(core_tests PRIVATE cpp/include)
set_target_properties(core_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(core_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME core_tests COMMAND core_tests)
